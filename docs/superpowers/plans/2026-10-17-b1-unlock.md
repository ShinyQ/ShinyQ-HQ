# B1 Engine Room, PR 3: The Unlock Implementation Plan

Status: planned (PR 3 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Visitors unlock the B1 Engine Room by reaching all five public floors: the store records the floors they reached and persists `b1Unlocked`; every floor picker shows a locked B1 row with a padlock and "Visit all 5 floors (n/5)" until then; the fifth floor plays a one-time unlock moment (chime, the row lights up, Rover offers "Take me there" or "Later"); `/{locale}#B1` and Cmd-K reach B1 only when unlocked; Ask the HQ mentions B1 in its site-build note with two new pills.

**Architecture:** Built on PR 2 (`2026-10-17-b1-engine-room-floor.md`, stacked under this branch): `b1Unlocked`, `unlockB1()`, `canEnterFloor`, `floorRows`, `towerFloors`, the B1 chunk and its prefetch rules already exist. This PR adds pure visit tracking in `src/experience/b1.ts` (`withFloorSeen`, `floorProgress`, `watchFloorVisits`, `b1FromHash`) behind two persisted store keys (`floorsSeen`, `b1Unlocked`; the persist `merge` fills them for older payloads, so `STORE_VERSION` stays 3), turns `floorRows` into "B1 always listed, locked until unlocked", and adds one HUD layer (`EngineUnlockLayer`) for the moment and the locked deep link. Locked visitors still never load or draw B1.

**Tech Stack:** Next.js 16 static export, React 19, React Three Fiber, zustand (persist), next-intl, Vitest, Playwright, Cloudflare Pages Functions (Ask), Bun.

## Decisions made in this plan (flag to the controller)

- Floors count as reached when the rover is in `explore` on them with no ride (arrival, deep link, resume, the Lobby after the intro). - `floorsSeen` and `b1Unlocked` are persisted in `hq:v1` without a version bump: the persist `merge` fills missing keys and drops invalid ones. The PR 2 dev flag now persists too (fresh test contexts are unaffected).
- The locked row shows the code "B1", a padlock and the hint; the floor name stays hidden until unlocked; the unlocked row reads "B1 Build" with the name "Engine Room" (the spec's "B1 Engine Room"). The pad picker's locked option has no thumbnail.

- Cmd-K: an action row "Go to B1 Engine Room" (EN) / "Ke Ruang Mesin B1" (ID): in 3D it rides to B1 when unlocked and toasts the hint when locked; on the Classic site it opens `/build`.

```ts
import type { HQState, HQStore } from "@/store/useHQStore";
import { isPublicFloor, PUBLIC_FLOORS } from "@/lib/floors";

/** Floors a visitor reaches to unlock B1: the five public floors (spec: "visited all 5 floors"). */
export const PUBLIC_FLOOR_COUNT = PUBLIC_FLOORS.length;

/** "n/5": public floors reached so far (B1 never counts). */
export function floorProgress(seen: readonly FloorId[]): { seen: number; total: number } {
  return { seen: PUBLIC_FLOORS.filter((f) => seen.includes(f)).length, total: PUBLIC_FLOOR_COUNT };
}

/**
 * The store patch for reaching `floor`: null when nothing changes (B1, or a floor already counted). The fifth
 * public floor unlocks B1 and starts the one-time unlock moment, unless B1 was already unlocked.
 */
export function withFloorSeen(state: { floorsSeen: readonly FloorId[]; b1Unlocked: boolean }, floor: FloorId): { floorsSeen: FloorId[]; b1Unlocked?: true; b1Moment?: true } | null {
  if (!isPublicFloor(floor) || state.floorsSeen.includes(floor)) return null;
  const floorsSeen = [...state.floorsSeen, floor];
  const all = PUBLIC_FLOORS.every((f) => floorsSeen.includes(f));
  return all && !state.b1Unlocked ? { floorsSeen, b1Unlocked: true, b1Moment: true } : { floorsSeen };
}

/** Records every floor the rover explores (arrival, deep link, resume, the Lobby after the intro); returns the unsubscribe. */
export function watchFloorVisits(store: HQStore): () => void {
  const mark = (s: HQState) => {
    if (s.phase === "explore" && s.ride === null) s.markFloorSeen(s.floor);
  };
  mark(store.getState());
  return store.subscribe(mark);
}

/** `/{locale}#B1` (any case): a deep link into the Engine Room from the Lobby route; other routes ignore it. */
export function b1FromHash(startFloor: FloorId | undefined, hash: string): "B1" | undefined {
  return !startFloor && /^#b1$/i.test(hash) ? "B1" : undefined;
}
```

export function scheduleChime(ctx: BaseAudioContext, destination: AudioNode, when: number): void {
  [1046.5, 1318.51, 1567.98].forEach((frequency, i) => {
    const start = when + i * 0.09;
    const end = start + 0.6;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = "sine";
    osc.frequency.setValueAtTime(frequency, start);
    envelope(gain, start, 0.12, 0.006, end);
    osc.connect(gain).connect(destination);
    cleanupOnEnded(osc, [osc, gain]);
    osc.start(start);
    osc.stop(end + 0.01);
  });
}

```tsx
"use client";

import { useTranslations } from "next-intl";
import { useEffect } from "react";
import { b1FromHash, floorProgress } from "@/experience/b1";
import { audio } from "@/lib/audio";
import { getHQStore, useHQStore } from "@/store/useHQStore";
import { useOnboardingSettled } from "./hooks";
import * as dialog from "./store";

const MOMENT = "b1-unlock";

/**
 * B1 Engine Room unlock (spec section 2): once the fifth public floor is reached (store `b1Moment`), a chime and
 * Rover's offer, "Take me there" (rides to B1) or "Later"; the elevator rows pulse meanwhile (FloorButtons). A
 * cancelled line comes back at the next chance; a ride or the tour first waits. A locked visitor who followed
 * `/{locale}#B1` hears how to unlock it. Renders nothing.
 */
export function EngineUnlockLayer() {
  const t = useTranslations("dialog.b1");
  const moment = useHQStore((s) => s.b1Moment);
  const free = useHQStore((s) => s.phase === "explore" && s.ride === null);
  const settled = useOnboardingSettled();

  useEffect(() => {
    if (!b1FromHash(undefined, window.location.hash)) return;
    window.history.replaceState(window.history.state, "", `${window.location.pathname}${window.location.search}`);
    const s = getHQStore().getState();
    if (s.b1Unlocked) return;
    const { seen, total } = floorProgress(s.floorsSeen);
    void dialog.say({ key: "b1-locked", text: t("locked", { seen, total }), face: "thinking" }, { id: "b1-locked" });
  }, [t]);

  useEffect(() => {
    if (!moment || !free || !settled) return;
    let live = true;
    audio.play("chime");
    void dialog
      .say({ key: MOMENT, text: t("unlock"), face: "arrived", choices: [{ id: "go", label: t("go") }, { id: "later", label: t("later"), cancel: true }] }, { id: MOMENT })
      .then((result) => {
        if (!live || result.status === "cancelled") return;
        const s = getHQStore().getState();
        s.clearB1Moment();
        if (result.choice === "go") s.requestElevator("B1");
      });
    return () => {
      live = false;
      dialog.cancelDialog(MOMENT);
    };
  }, [moment, free, settled, t]);

  return null;
}

/** Writes `floorsSeen` and `b1Unlocked` into the persisted store (`hq:v1`) before the first load of the tab only (a reload keeps what the session recorded). */
export async function seedFloors(page: Page, floors: string[], unlocked = false) {
  await page.addInitScript(
    ({ floors, unlocked }) => {
      if (sessionStorage.getItem("hq:test-seeded")) return;
      sessionStorage.setItem("hq:test-seeded", "1");
      const raw = localStorage.getItem("hq:v1");
      const saved = raw ? JSON.parse(raw) : { state: { visited: [], firstVisit: false, locale: "en", motion: "full" }, version: 3 };
      localStorage.setItem("hq:v1", JSON.stringify({ ...saved, state: { ...saved.state, floorsSeen: floors, b1Unlocked: unlocked } }));
    },
    { floors, unlocked },
  );
}
```
