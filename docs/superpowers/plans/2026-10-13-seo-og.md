```ts
#!/usr/bin/env bun
/**
 * Pixel diff for before/after screenshot gates (round 5: the aesthetic must not change). Reports the
 * share of pixels whose largest RGB channel difference exceeds --tolerance (default 24 of 255),
 * optionally inside --crop x,y,w,h only, and exits 1 above --max.
 *
 *   bun scripts/png-diff.ts before.png after.png --max 0.001 [--crop 0,0,1440,300] [--out mask.png]
 */
import sharp from "sharp";

export interface Crop {
  left: number;
  top: number;
  width: number;
  height: number;
}

export async function diffRatio(a: string | Buffer, b: string | Buffer, opts: { tolerance?: number; crop?: Crop } = {}) {
  const load = (src: string | Buffer) => {
    let image = sharp(src).removeAlpha().toColourspace("srgb");
    if (opts.crop) image = image.extract(opts.crop);
    return image.raw().toBuffer({ resolveWithObject: true });
  };
  const [x, y] = await Promise.all([load(a), load(b)]);
  if (x.info.width !== y.info.width || x.info.height !== y.info.height) {
    throw new Error(`size mismatch: ${x.info.width}x${x.info.height} vs ${y.info.width}x${y.info.height}`);
  }
  const tolerance = opts.tolerance ?? 24;
  const { width, height, channels } = x.info;
  const mask = Buffer.alloc(width * height);
  let changed = 0;
  for (let p = 0, i = 0; p < width * height; p++, i += channels) {
    let d = 0;
    for (let c = 0; c < 3; c++) d = Math.max(d, Math.abs(x.data[i + c] - y.data[i + c]));
    if (d > tolerance) {
      changed++;
      mask[p] = 255;
    }
  }
  return { ratio: changed / (width * height), width, height, mask };
}

if (import.meta.main) {
  const args = process.argv.slice(2);
  const flag = (name: string) => (args.includes(name) ? args[args.indexOf(name) + 1] : undefined);
  const [before, after] = args.filter((a, i) => !a.startsWith("--") && !args[i - 1]?.startsWith("--"));
  if (!before || !after) throw new Error("usage: bun scripts/png-diff.ts <before> <after> --max <ratio> [--crop x,y,w,h] [--out mask.png]");
  const max = Number(flag("--max") ?? "0");
  const cropArg = flag("--crop")?.split(",").map(Number);
  const crop = cropArg ? { left: cropArg[0], top: cropArg[1], width: cropArg[2], height: cropArg[3] } : undefined;
  const result = await diffRatio(before, after, { crop });
  const out = flag("--out");
  if (out) await sharp(result.mask, { raw: { width: result.width, height: result.height, channels: 1 } }).png().toFile(out);
  console.log(`${after}: ${(result.ratio * 100).toFixed(3)}% changed (max ${(max * 100).toFixed(3)}%)`);
  if (result.ratio > max) process.exit(1);
}
```

```ts
#!/usr/bin/env bun
/**
 * `og:stills` step 2: converts the "og still" captures (e2e/screenshots.spec.ts, 2400x1260 PNGs in
 * screenshots/og-stills/) into the committed OG backgrounds public/og-assets/{still}.jpg (1200x630,
 * JPEG q86, no metadata) and, with --photo <png>, writes the owner's photo as
 * public/og-assets/kurniadi-photo.png (400x400). The build reads only these files, so it never
 * needs WebGL. Run after `bun run build`: `E2E_PORT=4438 bun run og:stills [--photo path/to/photo.png]`.
 */
import { existsSync, mkdirSync, statSync } from "node:fs";
import path from "node:path";
import sharp from "sharp";
import { OG_PHOTO_PATH, OG_STILLS, ogStillPath } from "../src/lib/og-layout";

const SOURCE = path.join("screenshots", "og-stills");
const toPublic = (publicPath: string) => path.join("public", publicPath.replace(/^\//, ""));

mkdirSync(path.join("public", "og-assets"), { recursive: true });
for (const still of OG_STILLS) {
  const png = path.join(SOURCE, `${still}.png`);
  if (!existsSync(png)) throw new Error(`missing ${png}: run the "og still" captures first (bun run og:stills)`);
  const { channels } = await sharp(png).stats();
  const luma = 0.2126 * channels[0].mean + 0.7152 * channels[1].mean + 0.0722 * channels[2].mean;
  if (luma < 6) throw new Error(`${png} looks blank (mean luma ${luma.toFixed(1)}): the scene did not render`);
  const out = toPublic(ogStillPath(still));
  await sharp(png).resize(1200, 630, { fit: "cover", kernel: "lanczos3" }).jpeg({ quality: 86, mozjpeg: true, chromaSubsampling: "4:4:4" }).toFile(out);
  console.log(`${out} ${(statSync(out).size / 1024).toFixed(1)} KB (mean luma ${luma.toFixed(1)})`);
}

const photoArg = process.argv.indexOf("--photo");
if (photoArg > 0) {
  const out = toPublic(OG_PHOTO_PATH);
  await sharp(process.argv[photoArg + 1]).resize(400, 400, { fit: "cover" }).png({ compressionLevel: 9 }).toFile(out);
  console.log(`${out} ${(statSync(out).size / 1024).toFixed(1)} KB`);
}
if (!existsSync(toPublic(OG_PHOTO_PATH))) throw new Error(`missing public${OG_PHOTO_PATH}: pass --photo <png> once`);
```

```ts
/**
 * Satori renders PNG; a photographic card is about 1 MB as PNG and about 110 KB as JPEG, so the
 * route re-encodes to JPEG (4:4:4 chroma keeps neon text edges clean) and steps the quality down
 * until the card fits the share budget (300 KB, with headroom).
 */
import sharp from "sharp";

export const OG_JPEG_TARGET_BYTES = 290 * 1024;
const QUALITIES = [84, 80, 76, 72] as const;

export async function encodeOgJpeg(png: Buffer): Promise<Buffer> {
  let jpeg = Buffer.alloc(0);
  for (const quality of QUALITIES) {
    jpeg = await sharp(png).flatten({ background: "#05050c" }).jpeg({ quality, mozjpeg: true, chromaSubsampling: "4:4:4", progressive: true }).toBuffer();
    if (jpeg.length <= OG_JPEG_TARGET_BYTES) return jpeg;
  }
  return jpeg;
}
```

```ts
/**
 * Search and share copy for every page (owner request 2026-10-13: "fix the wording for the SEO").
 * Titles lead with the owner's name and value; descriptions are outcome-focused and 140 to 160
 * characters, built from content (taglines, key results, stack, excerpts, career summaries) and the
 * `seo` messages. Facts come from content/site-content.json, never from this file.
 */
import { createTranslator } from "next-intl";
import { getContent, getPod, getPost, getProfile, getTimeline, getTimelineEntry } from "@/content/load";
import type { Locale, Pod, PostRef, Result, TimelineEntry } from "@/content/schema";
import { hasFirstPerson } from "@/lib/first-person";
import type { OgPage, OgTarget } from "@/lib/site";
import en from "../../messages/en.json";
import id from "../../messages/id.json";

export const TITLE_MAX = 60;
export const HOME_TITLE_MAX = 95;
export const DYNAMIC_TITLE_MAX = 100;
export const DESCRIPTION_MIN = 140;
export const DESCRIPTION_MAX = 160;
/** Journey entry titles add the organization only while they stay this short. */
const ENTRY_TITLE_WITH_ORG_MAX = 70;

const MESSAGES = { en, id } as const;
const seoT = (locale: Locale) => createTranslator({ locale, messages: MESSAGES[locale], namespace: "seo" });

/** The result a card or description leads with: the first numeric result whose label is not first person. */
export function keyResult(pod: Pick<Pod, "results">): Result {
  const clean = pod.results.filter((r) => !hasFirstPerson(r.label.en, "en") && !hasFirstPerson(r.label.id, "id"));
  const pool = clean.length > 0 ? clean : pod.results;
  return pool.find((r) => /\d/.test(r.value)) ?? pool[0];
}

/** First and last year on the career timeline, e.g. { from: "2016", to: "2026" }. */
export function journeyYears(): { from: string; to: string } {
  const years = getTimeline()
    .flatMap((e) => [e.start, e.end])
    .filter((v) => v !== "present")
    .map((v) => Number(v.slice(0, 4)));
  return { from: String(Math.min(...years)), to: String(Math.max(...years)) };
}

/** "Caller interruption latency" -> "caller interruption latency"; acronyms ("AI triage") stay. */
export function lowerFirst(text: string): string {
  return /^[A-Z][A-Z]/.test(text) ? text : text.charAt(0).toLowerCase() + text.slice(1);
}

export function withStop(text: string): string {
  const t = text.trim();
  return /[.!?…]$/.test(t) ? t : `${t}.`;
}

/** Fits a lead into `max` characters: whole sentences first, then a clause, then words with an ellipsis. */
export function shortenLead(text: string, max = DESCRIPTION_MAX): string {
  const t = withStop(text);
  if (t.length <= max) return t;
  let out = "";
  for (const sentence of t.split(/(?<=[.!?])\s+/)) {
    const next = out ? `${out} ${sentence}` : sentence;
    if (next.length > max) break;
    out = next;
  }
  if (out.length >= 60) return out;
  const cut = t.slice(0, max - 1);
  for (const separator of ["; ", ": ", ", "]) {
    const i = cut.lastIndexOf(separator);
    if (i >= 60) return `${cut.slice(0, i)}.`;
  }
  return `${cut.slice(0, cut.lastIndexOf(" ")).replace(/[,;:]+$/, "")}…`;
}

function compareKeys(a: readonly number[], b: readonly number[]): number {
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return a[i] - b[i];
  return 0;
}

/**
 * The lead plus at most one option per group, within [min, max] characters. Prefers including
 * earlier groups, then earlier options in a group. Falls back to the longest text within `max`.
 */
export function fitDescription(lead: string, groups: readonly (readonly string[])[], min = DESCRIPTION_MIN, max = DESCRIPTION_MAX): string {
  const base = shortenLead(lead, max);
  let best: { text: string; key: number[] } | undefined;
  let longest = base;
  const walk = (i: number, parts: string[], included: number[], picks: number[]) => {
    if (i === groups.length) {
      const text = [base, ...parts].join(" ");
      if (text.length > max) return;
      if (text.length > longest.length) longest = text;
      const key = [...included, ...picks.map((j) => -j)];
      if (text.length >= min && (!best || compareKeys(key, best.key) > 0)) best = { text, key };
      return;
    }
    walk(i + 1, parts, [...included, 0], [...picks, 0]);
    groups[i].forEach((option, j) => walk(i + 1, [...parts, option], [...included, 1], [...picks, j]));
  };
  walk(0, [], [], []);
  return best?.text ?? longest;
}

/** "2022 to 2023", "2026 hingga sekarang", or one year when start and end share it. */
export function periodLabel(start: string, end: string, locale: Locale): string {
  const t = seoT(locale);
  const from = start.slice(0, 4);
  if (end !== "present" && end.slice(0, 4) === from) return from;
  return t("period", { from, to: end === "present" ? t("present") : end.slice(0, 4) });
}

  const title = t("postTitle", { title: post.title[locale], name });
  const description = fitDescription(post.excerpt[locale], [[t("postBy", { name }), t("articleBy", { name }), t("byName", { name })], [t("readArticle")]]);
  return { title, description, image: { kind: "blog", locale, slug: post.slug }, imageAlt: title, type: "article", publishedTime: post.date, tags: post.tags };
}

function entrySeo(entry: TimelineEntry, locale: Locale, name: string): SeoPage {
  const t = seoT(locale);
  const role = entry.role[locale];
  const period = periodLabel(entry.start, entry.end, locale);
  const withOrg = t("entryTitleOrg", { name, role, org: entry.org.replace(/\s*\([^)]*\)/g, "").trim() });
  const title = withOrg.length <= ENTRY_TITLE_WITH_ORG_MAX ? withOrg : t("entryTitle", { name, role });
  const description = fitDescription(entry.summary[locale], [
    [t("entryRole", { role, org: entry.org, period }), t("entryOrg", { org: entry.org, period }), t("entryPeriod", { period })],
    [t("careerOf", { name }), t("byName", { name })],
    [t("seeJourney")],
  ]);
  return { title, description, image: { kind: "page", locale, page: "journey" }, imageAlt: title, type: "website" };
}

/** Title, description and share card of a locale-relative page path from `allPagePaths()`. */
export function seoFor(locale: Locale, path: string): SeoPage {
  const t = seoT(locale);
  const profile = getProfile();
  const name = profile.name;
  const values = {
    name,
    city: profile.location[locale].split(",")[0].trim(),
    motto: profile.subheadline[locale],
    count: String(getContent().floors.labs.pods.length),
    ...journeyYears(),
  };
  if (path === "/") {
    const title = t("homeTitle", values);
    return { title, description: t("homeDescription", values), image: { kind: "default", locale }, imageAlt: title, type: "website" };
  }
  const page = SECTION_PAGES[path];
  if (page) {
    const title = t(`${page}Title`, values);
    return { title, description: t(`${page}Description`, values), image: { kind: "page", locale, page }, imageAlt: title, type: "website" };
  }
  const [, section, slug] = path.split("/");
  if (section === "labs") {
    const pod = getPod(slug);
    if (pod) return podSeo(pod, locale, name);
  }
  if (section === "blog") {
    const post = getPost(slug);
    if (post && !post.url) return postSeo(post, locale, name);
  }
  if (section === "journey") {
    const entry = getTimelineEntry(slug);
    if (entry) return entrySeo(entry, locale, name);
  }
  throw new Error(`No search metadata for ${path}`);
}
```

```ts
/**
 * Pure helpers for the social-card check (scripts/og-check.ts and e2e/seo.spec.ts): read a page's
 * search and share tags from its HTML, validate them, and read a JPEG's dimensions. No `@/`
 * imports, so Playwright specs can import this file directly.
 */
export const DESCRIPTION_RANGE = [140, 160] as const;
export const OG_MAX_BYTES = 300 * 1024;

export interface SocialMeta {
  title?: string;
  description?: string;
  canonical?: string;
  hreflang: Record<string, string>;
  og: Record<string, string>;
  twitter: Record<string, string>;
}

const decode = (s: string) =>
  s.replace(/&quot;/g, '"').replace(/&#x27;|&#39;/g, "'").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&amp;/g, "&");

function attributes(tag: string): Record<string, string> {
  const out: Record<string, string> = {};
  for (const m of tag.matchAll(/([a-zA-Z:-]+)="([^"]*)"/g)) out[m[1].toLowerCase()] = decode(m[2]);
  return out;
}

export function parseSocialMeta(html: string): SocialMeta {
  const end = html.indexOf("</head>");
  const head = end === -1 ? html : html.slice(0, end);
  const meta: SocialMeta = { hreflang: {}, og: {}, twitter: {} };
  const title = head.match(/<title>([^<]*)<\/title>/);
  if (title) meta.title = decode(title[1]);
  for (const m of head.matchAll(/<meta\s[^>]*>/g)) {
    const a = attributes(m[0]);
    const key = a.property ?? a.name;
    if (!key || a.content === undefined) continue;
    if (key === "description") meta.description = a.content;
    else if (key.startsWith("og:")) meta.og[key] ??= a.content;
    else if (key.startsWith("twitter:")) meta.twitter[key] ??= a.content;
  }
  for (const m of head.matchAll(/<link\s[^>]*>/g)) {
    const a = attributes(m[0]);
    if (a.rel === "canonical") meta.canonical = a.href;
    else if (a.rel === "alternate" && a.hreflang) meta.hreflang[a.hreflang] = a.href;
  }
  return meta;
}

export function validateSocialMeta(meta: SocialMeta): string[] {
  const problems: string[] = [];
  const need = (ok: unknown, problem: string) => {
    if (!ok) problems.push(problem);
  };
  need(meta.title, "missing <title>");
  const length = meta.description?.length ?? 0;
  need(length >= DESCRIPTION_RANGE[0] && length <= DESCRIPTION_RANGE[1], `description length ${length} outside ${DESCRIPTION_RANGE[0]} to ${DESCRIPTION_RANGE[1]}`);
  need(meta.canonical, "missing canonical");
  for (const lang of ["en", "id", "x-default"]) need(meta.hreflang[lang], `missing hreflang ${lang}`);
  for (const key of ["og:title", "og:description", "og:url", "og:image", "og:image:alt"]) need(meta.og[key], `missing ${key}`);
  need(meta.og["og:image:width"] === "1200" && meta.og["og:image:height"] === "630", "og:image is not declared 1200x630");
  need(meta.og["og:image:type"] === "image/jpeg", "og:image:type is not image/jpeg");
  need(meta.twitter["twitter:card"] === "summary_large_image", "twitter:card is not summary_large_image");
  need(meta.twitter["twitter:image"], "missing twitter:image");
  return problems;
}

/** Width and height from a JPEG's SOF marker (baseline or progressive), or null when the bytes are not a JPEG. */
export function jpegSize(bytes: Uint8Array): { width: number; height: number } | null {
  if (bytes[0] !== 0xff || bytes[1] !== 0xd8) return null;
  let i = 2;
  while (i + 8 < bytes.length) {
    if (bytes[i] !== 0xff) {
      i++;
      continue;
    }
    const marker = bytes[i + 1];
    if (marker === 0xff || marker === 0x01 || (marker >= 0xd0 && marker <= 0xd8)) {
      i += marker === 0xff ? 1 : 2;
      continue;
    }
    if (marker >= 0xc0 && marker <= 0xcf && marker !== 0xc4 && marker !== 0xc8 && marker !== 0xcc) {
      return { height: (bytes[i + 5] << 8) | bytes[i + 6], width: (bytes[i + 7] << 8) | bytes[i + 8] };
    }
    i += 2 + ((bytes[i + 2] << 8) | bytes[i + 3]);
  }
  return null;
}

/** The exported HTML of a page: ("out", "en", "/") -> "out/en.html", "/labs/x" -> "out/en/labs/x.html". */
export function exportHtmlPath(outDir: string, locale: string, pagePath: string): string {
  return `${outDir}/${locale}${pagePath === "/" ? "" : pagePath}.html`;
}
```

```ts
/**
 * Social-card preview check over the static export (run after `bun run build`; part of
 * `verify:quick`). Every page in both locales plus the root redirect page must have a title, a
 * 140 to 160 character description, a canonical, hreflang en/id/x-default, og:title, og:description,

 * 300 KB. Writes 300 px previews and a contact sheet (screenshots/og-preview/index.html) for the

 *
 *   bun scripts/og-check.ts [--out out] [--previews screenshots/og-preview]

import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import sharp from "sharp";
import { LOCALES } from "@/content/schema";
import { allPagePaths } from "@/lib/routes";
import { OG_MAX_BYTES, exportHtmlPath, jpegSize, parseSocialMeta, validateSocialMeta } from "./og-meta";

const args = process.argv.slice(2);
const flag = (name: string, fallback: string) => (args.includes(name) ? args[args.indexOf(name) + 1] : fallback);
const outDir = flag("--out", "out");
const previewDir = flag("--previews", path.join("screenshots", "og-preview"));

const problems: string[] = [];
const images = new Map<string, string[]>();
const pages = [
  { file: path.join(outDir, "index.html"), label: "/" },
  ...LOCALES.flatMap((locale) => allPagePaths().map((p) => ({ file: exportHtmlPath(outDir, locale, p), label: `/${locale}${p === "/" ? "" : p}` }))),
];

for (const page of pages) {
  if (!existsSync(page.file)) {
    problems.push(`${page.label}: missing ${page.file}`);
    continue;
  }
  const meta = parseSocialMeta(readFileSync(page.file, "utf8"));
  for (const problem of validateSocialMeta(meta)) problems.push(`${page.label}: ${problem}`);
  const url = meta.og["og:image"];
  if (url) {
    const rel = new URL(url, "https://placeholder.invalid").pathname;
    images.set(rel, [...(images.get(rel) ?? []), page.label]);
  }
}

mkdirSync(previewDir, { recursive: true });
const rows: string[] = [];
const sizes: number[] = [];
for (const [rel, usedBy] of images) {
  const file = path.join(outDir, rel);
  if (!existsSync(file)) {
    problems.push(`${rel}: image missing (used by ${usedBy[0]})`);
    continue;
  }
  const bytes = readFileSync(file);
  sizes.push(bytes.length);
  const size = jpegSize(bytes);
  if (!size || size.width !== 1200 || size.height !== 630) problems.push(`${rel}: not a 1200x630 JPEG`);
  if (bytes.length > OG_MAX_BYTES) problems.push(`${rel}: ${(bytes.length / 1024).toFixed(0)} KB, over ${OG_MAX_BYTES / 1024} KB`);
  const name = rel.replace(/^\/og\//, "").replace(/\//g, "_").replace(/\.jpg$/, "");
  await sharp(bytes).resize(300).png().toFile(path.join(previewDir, `${name}-300.png`));
  rows.push(
    `<figure><img src="${path.relative(previewDir, file)}" width="600"><img src="${name}-300.png" width="300"><figcaption>${rel}<br>${(bytes.length / 1024).toFixed(0)} KB, ${usedBy.length} page(s)</figcaption></figure>`,
  );
}

writeFileSync(
  path.join(previewDir, "index.html"),
  `<!doctype html><meta charset="utf-8"><title>OG preview</title><style>body{background:#111;color:#ddd;font:14px system-ui}figure{display:flex;gap:16px;align-items:flex-start;margin:24px}</style>\n${rows.join("\n")}\n`,
);
console.log(
  `${pages.length} pages, ${images.size} share images, largest ${(Math.max(...sizes) / 1024).toFixed(0)} KB, total ${(sizes.reduce((a, b) => a + b, 0) / 1024 / 1024).toFixed(1)} MB. Previews: ${previewDir}/index.html`,
);
if (problems.length > 0) {
  console.error(problems.join("\n"));
  process.exit(1);
}
```
